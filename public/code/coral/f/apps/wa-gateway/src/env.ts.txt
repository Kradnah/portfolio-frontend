import { tmpdir } from 'node:os';
import { join } from 'node:path';

export interface Env {
  REDIS_URL: string;
  /** Chromium del sistema (en la imagen: /usr/bin/chromium). */
  CHROME_PATH: string | undefined;
  /** Carpetas de autenticación de WhatsApp Web; se borran al cerrar cada sesión. */
  SESSIONS_DIR: string;
}

export function loadEnv(source: NodeJS.ProcessEnv = process.env): Env {
  const redis = source.REDIS_URL;
  if (!redis || !/^rediss?:\/\//.test(redis)) throw new Error('REDIS_URL es obligatoria');
  return {
    REDIS_URL: redis,
    CHROME_PATH: source.CHROME_PATH || undefined,
    SESSIONS_DIR: source.SESSIONS_DIR || join(tmpdir(), 'coral-wa'),
  };
}

/** Conexión de BullMQ (exige `maxRetriesPerRequest: null`). */
export function redisConnection(url: string) {
  const u = new URL(url);
  return {
    host: u.hostname,
    port: Number(u.port || 6379),
    username: u.username || undefined,
    password: u.password ? decodeURIComponent(u.password) : undefined,
    db: u.pathname.length > 1 ? Number(u.pathname.slice(1)) : 0,
    tls: u.protocol === 'rediss:' ? {} : undefined,
    maxRetriesPerRequest: null,
  };
}
