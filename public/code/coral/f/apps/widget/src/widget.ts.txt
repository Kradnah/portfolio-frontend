// Con `data-demo`, la primera vez que se abre crea la demo del visitante (sandbox de 4 h) y ofrece
// "Ver en el CRM" para abrir esa misma conversación en el panel. Shadow DOM: no hereda ni filtra
// estilos de la página que lo aloja.

  id: string;
  sender: 'CONTACT' | 'BOT' | 'AGENT' | 'SYSTEM';
  text: string;
}

interface Stored {
  widgetKey: string;
  claimToken: string;
  sandboxExpiresAt: string;
  visitorId: string;
}

class CoralChat {
  private readonly root: ShadowRoot;
  private readonly t: (typeof TEXT)['es'];
  private socket?: Socket;
  private stored: Stored | null = null;
  private readonly el: Record<string, HTMLElement> = {};

  constructor(
    host: HTMLElement,
    private readonly api: string,
    lang: 'es' | 'en',
  ) {
    this.t = TEXT[lang];
    this.root = host.attachShadow({ mode: 'open' });
    this.root.innerHTML = `
      <style>${STYLES}</style>
      <button class="launcher" type="button"><span class="dot" aria-hidden="true"></span><span></span></button>
      <section class="panel" role="dialog" aria-modal="false" hidden>
        <div class="stripe" aria-hidden="true"></div>
        <div class="head"><strong></strong><a class="crm" target="_blank" rel="noopener" aria-disabled="true"></a>
          <button class="x" type="button">×</button></div>
        <p class="note"></p>
        <div class="list" aria-live="polite"></div>
        <p class="status" hidden></p>
        <form class="form"><input maxlength="2000" autocomplete="off" /><button type="submit"></button></form>
      </section>`;
    const q = <T extends HTMLElement>(s: string) => this.root.querySelector(s) as T;
    Object.assign(this.el, {
      launcher: q('.launcher'),
      panel: q('.panel'),
      crm: q('.crm'),
      list: q('.list'),
      status: q('.status'),
      form: q('.form'),
      input: q('.form input'),
      send: q('.form button'),
    });
    (q('.launcher span:last-child') as HTMLElement).textContent = this.t.launcher;
    (q('.head strong') as HTMLElement).textContent = this.t.title;
    this.el['crm']!.textContent = this.t.crm;
    (q('.x') as HTMLElement).setAttribute('aria-label', this.t.close);
    (q('.note') as HTMLElement).textContent = this.t.note;
    (this.el['input'] as HTMLInputElement).placeholder = this.t.placeholder;
    this.el['send']!.textContent = this.t.send;
    this.el['panel']!.setAttribute('aria-label', this.t.title);

    this.el['launcher']!.addEventListener('click', () => void this.open());
    q('.x').addEventListener('click', () => this.close());
    this.el['form']!.addEventListener('submit', (e) => {
      e.preventDefault();
      void this.send();
    });
  }

  private async open(): Promise<void> {
    this.el['launcher']!.hidden = true;
    this.el['panel']!.hidden = false;
    (this.el['input'] as HTMLInputElement).focus();
    if (this.socket) return;
    try {
      this.stored = await this.session();
    } catch (err) {
      this.showStatus((err as Error).message === 'limit' ? this.t.limit : this.t.error);
      return;
    }
    const crm = this.el['crm'] as HTMLAnchorElement;
    crm.href = `${this.api}/demo/claim?t=${encodeURIComponent(this.stored.claimToken)}`;
    crm.removeAttribute('aria-disabled');
    this.connect();
  }

  private close(): void {
    this.el['panel']!.hidden = true;
    this.el['launcher']!.hidden = false;
  }

  /** La demo del visitante: la guardada en este navegador si sigue viva, o una nueva. */
  private async session(): Promise<Stored> {
    try {
      const saved = JSON.parse(localStorage.getItem(STORAGE) ?? 'null') as Stored | null;
      if (saved && Date.parse(saved.sandboxExpiresAt) > Date.now() + 60_000) return saved;
    } catch {
      // Sin almacenamiento: se crea una demo por visita.
    }
    const res = await fetch(`${this.api}/api/demo/widget`, { method: 'POST' });
    if (res.status === 429) throw new Error('limit');
    if (!res.ok) throw new Error('error');
    const data = (await res.json()) as Omit<Stored, 'visitorId'>;
    const stored = { ...data, visitorId: crypto.randomUUID() };
    try {
      localStorage.setItem(STORAGE, JSON.stringify(stored));
    } catch {
      // Igual funciona durante esta visita.
    }
    return stored;
  }

  private connect(): void {
    const s = this.stored!;
    const socket = io(`${this.api}/widget`, {
      path: SOCKET_PATH,
      transports: ['websocket'],
      auth: { key: s.widgetKey, visitorId: s.visitorId, name: 'Visitante del portfolio' },
    });
    socket.on('connect', async () => {
      this.showStatus(null);
      const hello = (await socket.emitWithAck(WidgetEvent.Hello)) as { messages: Msg[] };
      this.el['list']!.textContent = '';
      if (!hello.messages.length) this.add({ id: 'hello', sender: 'BOT', text: this.t.hello });
      hello.messages.forEach((m) => this.add(m));
    });
    socket.on('connect_error', () => {
      // La demo pudo vencer: se olvida y la próxima apertura crea otra.
      try {
        localStorage.removeItem(STORAGE);
      } catch {
        /* nada */
      }
      this.showStatus(this.t.error);
    });
    socket.on(WidgetEvent.Typing, (on: boolean) => this.typing(on));
    socket.on(WidgetEvent.Message, (m: Msg) => {
      this.typing(false);
      this.add(m);
    });
    this.socket = socket;
  }

  private async send(): Promise<void> {
    const input = this.el['input'] as HTMLInputElement;
    const text = input.value.trim();
    if (!text || !this.socket?.connected) return;
    input.value = '';
    const res = (await this.socket.emitWithAck(WidgetEvent.Send, { text })) as {
      ok: boolean;
      id?: string;
      error?: string;
    };
    if (res.ok) this.add({ id: res.id!, sender: 'CONTACT', text });
    else this.showStatus(res.error === 'RATE_LIMITED' ? this.t.fast : this.t.error);
  }

  private add(m: Msg): void {
    const list = this.el['list']!;
    if (m.id && list.querySelector(`[data-id="${CSS.escape(m.id)}"]`)) return;
    const div = document.createElement('div');
    div.className = `m ${{ CONTACT: 'me', BOT: 'bot', AGENT: 'agent', SYSTEM: 'sys' }[m.sender]}`;
    div.dataset['id'] = m.id;
    div.textContent = m.text;
    list.querySelector('.typing')?.before(div) ?? list.append(div);
    list.scrollTop = list.scrollHeight;
  }

  private typing(on: boolean): void {
    const list = this.el['list']!;
    let el = list.querySelector('.typing');
    if (on && !el) {
      el = document.createElement('div');
      el.className = 'typing';
      el.textContent = this.t.typing;
      list.append(el);
      list.scrollTop = list.scrollHeight;
    } else if (!on) el?.remove();
  }

  private showStatus(text: string | null): void {
    const status = this.el['status']!;
    status.hidden = !text;
    status.textContent = text ?? '';
  }
}

// Arranque: un solo widget por página, configurado con los atributos del <script>.
(() => {
  const script = document.currentScript as HTMLScriptElement | null;
  if (!script || document.querySelector('coral-chat-host')) return;
  const api = script.dataset['api'] ?? new URL(script.src).origin;
  const lang = (script.dataset['lang'] ?? document.documentElement.lang ?? 'es').startsWith('en')
    ? 'en'
    : 'es';
  const host = document.createElement('coral-chat-host');
  document.body.append(host);
  new CoralChat(host, api, lang);
})();
