import { describe, expect, it } from 'vitest';
import { InvalidEnvError, parseEnv } from '../src/config/env.js';
import { BASE_ENV, testEnv } from './helpers/env.js';

describe('parseEnv', () => {
  it('acepta un entorno mínimo y aplica valores por defecto', () => {
    const env = testEnv();
    expect(env.PORT).toBe(3000);
    expect(env.ACCESS_TOKEN_TTL_SECONDS).toBe(900);
    expect(env.CORS_ORIGINS).toEqual([]);
  });

  it('rechaza una clave JWT que no es EC P-256 sin mostrar su valor', () => {
    try {
      parseEnv({ ...BASE_ENV, JWT_PRIVATE_KEY: 'no-es-una-clave' });
      expect.unreachable();
    } catch (err) {
      expect(err).toBeInstanceOf(InvalidEnvError);
      expect((err as Error).message).toContain('JWT_PRIVATE_KEY');
      expect((err as Error).message).not.toContain('no-es-una-clave');
    }
  });

  it('en producción exige orígenes CORS https', () => {
    expect(() =>
      parseEnv({ ...BASE_ENV, NODE_ENV: 'production', CORS_ORIGINS: 'http://example.com' }),
    ).toThrow(InvalidEnvError);
  });
});
