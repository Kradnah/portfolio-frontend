import { createPrivateKey } from 'node:crypto';
import { existsSync } from 'node:fs';
import { z } from 'zod';

/**
 * Clave PEM en la variable de entorno: admite el PEM tal cual, con saltos de línea escapados
 * (`\n` literal) o codificado en base64 (más cómodo en paneles como Dokploy).
 */
const pemKey = z
  .string()
  .min(1)
  .transform((v) => {
    const raw = v.trim();
    if (raw.startsWith('-----BEGIN')) return raw.replace(/\\n/g, '\n');
    return Buffer.from(raw, 'base64').toString('utf8');
  })
  .refine((pem) => {
    try {
      const key = createPrivateKey(pem);
      return (
        key.asymmetricKeyType === 'ec' && key.asymmetricKeyDetails?.namedCurve === 'prime256v1'
      );
    } catch {
      return false;
    }
  }, 'Debe ser una clave privada EC P-256 (ES256) en PEM o base64');

const booleanString = z
  .enum(['true', 'false', '1', '0'])
  .transform((v) => v === 'true' || v === '1');

const commaList = z
  .string()
  .default('')
  .transform((v) =>
    v
      .split(',')
      .map((s) => s.trim())
      .filter(Boolean),
  )
  .pipe(z.array(z.url()));

/** Variables de entorno del API. Se validan al arrancar: si falta algo obligatorio, no arranca. */
export const EnvSchema = z
  .object({
    NODE_ENV: z.enum(['development', 'test', 'production']).default('development'),
    PORT: z.coerce.number().int().min(1).max(65535).default(3000),
    HOST: z.string().default('127.0.0.1'),
    LOG_LEVEL: z.enum(['fatal', 'error', 'warn', 'info', 'debug', 'trace']).default('info'),
    /** true detrás de Traefik, para que la IP del cliente (rate limiting, sesiones) sea la real. */
    TRUST_PROXY: booleanString.default(false),

    DATABASE_URL: z
      .url()
      .refine((u) => /^postgres(ql)?:\/\//.test(u), 'Debe ser una URL de Postgres'),
    REDIS_URL: z.url().refine((u) => /^rediss?:\/\//.test(u), 'Debe ser una URL de Redis'),

    /** Orígenes con CORS (widget en sitios de terceros). El panel llama por su mismo dominio. */
    CORS_ORIGINS: commaList,

    /** Clave privada ES256 de los access tokens; la pública se deriva de ella. */
    JWT_PRIVATE_KEY: pemKey,
    /** Identificador de la clave (cabecera `kid`), para poder rotarla. */
    JWT_KID: z.string().regex(/^[A-Za-z0-9._-]{1,64}$/),
    /** OpenRouter (API compatible con OpenAI). Sin clave, el bot responde un mensaje fijo. */
    OPENROUTER_API_KEY: z.string().min(1).optional(),
    OPENROUTER_MODEL: z.string().default('google/gemini-2.5-flash-lite'),
    OPENROUTER_URL: z.url().default('https://openrouter.ai/api/v1/chat/completions'),
    /** Límites del agente: respuestas de IA por sandbox y costo diario total (USD). */
    AI_REPLIES_PER_SANDBOX: z.coerce.number().int().min(1).default(40),
    AI_DAILY_BUDGET_USD: z.coerce.number().positive().default(0.5),
    /** Espera tras el último mensaje del contacto antes de responder (agrupa mensajes seguidos). */
    AI_DEBOUNCE_MS: z.coerce.number().int().min(0).max(20_000).default(2_500),

    /** Demo pública: sandboxes por IP y en total por día. */
    DEMO_SANDBOX_PER_IP_DAILY: z.coerce.number().int().min(1).default(3),
    DEMO_SANDBOX_GLOBAL_DAILY: z.coerce.number().int().min(1).default(40),
    ACCESS_TOKEN_TTL_SECONDS: z.coerce.number().int().min(60).max(900).default(900),
    REFRESH_TOKEN_TTL_DAYS: z.coerce.number().int().min(1).max(90).default(30),
  })
  .superRefine((env, ctx) => {
    if (env.NODE_ENV === 'production' && env.CORS_ORIGINS.some((o) => !o.startsWith('https://'))) {
      ctx.addIssue({
        code: 'custom',
        path: ['CORS_ORIGINS'],
        message: 'En producción solo se permiten orígenes https://',
      });
    }
  });

export type Env = z.infer<typeof EnvSchema>;

export class InvalidEnvError extends Error {
  constructor(readonly issues: string[]) {
    super(`Configuración inválida:\n${issues.map((i) => `  - ${i}`).join('\n')}`);
  }
}

/** Valida un objeto de entorno. Los mensajes nunca incluyen los valores (pueden ser secretos). */
export function parseEnv(source: Record<string, string | undefined>): Env {
  const result = EnvSchema.safeParse(source);
  if (!result.success) {
    throw new InvalidEnvError(
      result.error.issues.map((i) => `${i.path.join('.') || '(raíz)'}: ${i.message}`),
    );
  }
  return result.data;
}

/** Carga apps/api/.env en desarrollo (si existe) y valida `process.env`. */
export function loadEnv(): Env {
  if (process.env['NODE_ENV'] !== 'production' && existsSync('.env')) {
    process.loadEnvFile('.env');
  }
  return parseEnv(process.env);
}
