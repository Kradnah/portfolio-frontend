import { createHash, randomBytes, randomUUID } from 'node:crypto';
import { Injectable, Logger } from '@nestjs/common';
import type { RequestContext } from '../../common/auth/auth.decorators.js';
import { InjectEnv } from '../../config/config.module.js';
import type { Env } from '../../config/env.js';
import { PrismaService } from '../../infra/prisma/prisma.service.js';
import { SessionRevocationService } from './session-revocation.service.js';

const DAY_MS = 24 * 60 * 60 * 1000;

export const hashToken = (raw: string) => createHash('sha256').update(raw).digest('hex');

export interface IssuedRefreshToken {
  raw: string;
  familyId: string;
  expiresAt: Date;
}

export type RotateResult =
  | {
      ok: true;
      userId: string;
      organizationId: string;
      refresh: IssuedRefreshToken;
    }
  | { ok: false; reason: 'INVALID' | 'EXPIRED' | 'REUSED' };

/**
 * Refresh tokens rotativos: 256 bits aleatorios, en BD solo su SHA-256, uno nuevo en cada uso.
 * Si llega uno ya rotado, se asume robo y se revoca toda la familia (la sesión).
 */
@Injectable()
export class SessionService {
  private readonly logger = new Logger(SessionService.name);
  private readonly ttlMs: number;

  constructor(
    private readonly prisma: PrismaService,
    private readonly revocations: SessionRevocationService,
    @InjectEnv() env: Env,
  ) {
    this.ttlMs = env.REFRESH_TOKEN_TTL_DAYS * DAY_MS;
  }

  /**
   * Abre una sesión nueva (familia nueva) en la organización indicada. `until` acota la sesión
   * (p. ej. al vencimiento de un sandbox de la demo).
   */
  async start(
    userId: string,
    organizationId: string,
    ctx: RequestContext,
    until?: Date,
  ): Promise<IssuedRefreshToken> {
    const raw = this.newToken();
    const familyId = randomUUID();
    const max = Date.now() + this.ttlMs;
    const expiresAt = new Date(until ? Math.min(until.getTime(), max) : max);
    await this.prisma.refreshToken.create({
      data: {
        userId,
        familyId,
        organizationId,
        tokenHash: hashToken(raw),
        userAgent: ctx.userAgent,
        ip: ctx.ip,
        expiresAt,
      },
    });
    return { raw, familyId, expiresAt };
  }

  /**
   * Rota el refresh token. `organizationId` permite cambiar la organización activa de la sesión
   * en el mismo paso (el llamador comprueba antes que el usuario pertenece a ella).
   */
  async rotate(raw: string, ctx: RequestContext, organizationId?: string): Promise<RotateResult> {
    const current = await this.prisma.refreshToken.findUnique({
      where: { tokenHash: hashToken(raw) },
    });
    if (!current) return { ok: false, reason: 'INVALID' };

    if (current.revokedAt) {
      await this.revokeFamily(current.familyId);
      this.logger.warn(
        { userId: current.userId, familyId: current.familyId },
        'Reutilización de refresh token',
      );
      return { ok: false, reason: 'REUSED' };
    }
    if (current.expiresAt.getTime() <= Date.now()) return { ok: false, reason: 'EXPIRED' };

    const next = this.newToken();
    const nextId = randomUUID();
    const expiresAt = new Date(Date.now() + this.ttlMs);
    const nextOrg = organizationId ?? current.organizationId;
    const rotated = await this.prisma.$transaction(async (tx) => {
      // Condición sobre revoked_at: si dos peticiones rotan el mismo token, solo una gana.
      const { count } = await tx.refreshToken.updateMany({
        where: { id: current.id, revokedAt: null },
        data: { revokedAt: new Date(), replacedBy: nextId },
      });
      if (count === 0) return false;
      await tx.refreshToken.create({
        data: {
          id: nextId,
          userId: current.userId,
          familyId: current.familyId,
          organizationId: nextOrg,
          tokenHash: hashToken(next),
          userAgent: ctx.userAgent,
          ip: ctx.ip,
          expiresAt,
        },
      });
      return true;
    });

    if (!rotated) {
      await this.revokeFamily(current.familyId);
      return { ok: false, reason: 'REUSED' };
    }
    return {
      ok: true,
      userId: current.userId,
      organizationId: nextOrg,
      refresh: { raw: next, familyId: current.familyId, expiresAt },
    };
  }

  /** Familia (sesión) a la que pertenece un refresh token, si existe. */
  async findFamily(raw: string): Promise<string | null> {
    const token = await this.prisma.refreshToken.findUnique({
      where: { tokenHash: hashToken(raw) },
      select: { familyId: true },
    });
    return token?.familyId ?? null;
  }

  async revokeFamily(familyId: string): Promise<void> {
    await this.prisma.refreshToken.updateMany({
      where: { familyId, revokedAt: null },
      data: { revokedAt: new Date() },
    });
    await this.revocations.revoke(familyId);
  }

  private newToken(): string {
    return randomBytes(32).toString('base64url');
  }
}
