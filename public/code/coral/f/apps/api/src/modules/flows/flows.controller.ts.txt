import { Body, Controller, Get, HttpCode, Post, Put } from '@nestjs/common';
import { type FlowDto, SaveFlowRequest } from '@coral/contracts';
import { type AuthUser, CurrentUser } from '../../common/auth/auth.decorators.js';
import { createZodDto } from '../../common/zod/zod-dto.js';
import { FlowsService } from './flows.service.js';

class SaveFlowDto extends createZodDto(SaveFlowRequest) {}

/** Árbol de decisiones de la organización activa. */
@Controller('flow')
export class FlowsController {
  constructor(private readonly flows: FlowsService) {}

  @Get()
  get(@CurrentUser() user: AuthUser): Promise<FlowDto> {
    return this.flows.get(user.organizationId);
  }

  @Put()
  save(@CurrentUser() user: AuthUser, @Body() body: SaveFlowDto): Promise<FlowDto> {
    return this.flows.save(user.organizationId, body);
  }

  /** Vuelve a la plantilla de Casa Zhef. */
  @HttpCode(200)
  @Post('reset')
  reset(@CurrentUser() user: AuthUser): Promise<FlowDto> {
    return this.flows.reset(user.organizationId);
  }
}
